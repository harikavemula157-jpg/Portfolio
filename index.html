          GradeCalculator.java 
          public class GradeCalculator {
    public static final int PASS_MARK = 35;

    public static double percentage(Student s) {
        return s.getTotal() / 3.0;
    }

    public static String grade(Student s) {
        if (!passed(s)) {
            return "F";
        }
        double p = percentage(s);
        if (p >= 90) return "A+";
        if (p >= 80) return "A";
        if (p >= 70) return "B";
        if (p >= 60) return "C";
        if (p >= 50) return "D";
        return "E";
    }

    /** A student passes only if every subject is at or above the pass mark. */
    public static boolean passed(Student s) {
        return s.getJavaMarks() >= PASS_MARK
            && s.getDbmsMarks() >= PASS_MARK
            && s.getCMarks() >= PASS_MARK;
    }
}
          Student.java 
          public class Student {
    private final String rollNo;
    private final String name;
    private final int javaMarks;
    private final int dbmsMarks;
    private final int cMarks;

    public Student(String rollNo, String name, int javaMarks, int dbmsMarks, int cMarks) {
        if (rollNo == null || rollNo.isBlank()) {
            throw new IllegalArgumentException("Roll number is required");
        }
        if (name == null || name.isBlank()) {
            throw new IllegalArgumentException("Name is required");
        }
        checkMarks(javaMarks);
        checkMarks(dbmsMarks);
        checkMarks(cMarks);
        this.rollNo = rollNo.trim();
        this.name = name.trim();
        this.javaMarks = javaMarks;
        this.dbmsMarks = dbmsMarks;
        this.cMarks = cMarks;
    }

    private static void checkMarks(int marks) {
        if (marks < 0 || marks > 100) {
            throw new IllegalArgumentException("Marks must be between 0 and 100, got " + marks);
        }
    }

    public String getRollNo() { return rollNo; }
    public String getName() { return name; }
    public int getJavaMarks() { return javaMarks; }
    public int getDbmsMarks() { return dbmsMarks; }
    public int getCMarks() { return cMarks; }

    public int getTotal() { return javaMarks + dbmsMarks + cMarks; }
}
          StudentDAO.java (JDBC and MySQL) 
          import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;
import java.util.ArrayList;
import java.util.List;

/** All database access lives here, using JDBC and prepared statements. */
public class StudentDAO {
    private final String url;
    private final String user;
    private final String password;

    public StudentDAO() {
        this.url = env("DB_URL", "jdbc:mysql://localhost:3306/grade_manager");
        this.user = env("DB_USER", "root");
        this.password = env("DB_PASS", "");
    }

    private static String env(String key, String fallback) {
        String v = System.getenv(key);
        return (v == null || v.isEmpty()) ? fallback : v;
    }

    private Connection connect() throws SQLException {
        return DriverManager.getConnection(url, user, password);
    }

    public void add(Student s) throws SQLException {
        String sql = "INSERT INTO students (roll_no, name, java_marks, dbms_marks, c_marks) VALUES (?, ?, ?, ?, ?)";
        try (Connection c = connect(); PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setString(1, s.getRollNo());
            ps.setString(2, s.getName());
            ps.setInt(3, s.getJavaMarks());
            ps.setInt(4, s.getDbmsMarks());
            ps.setInt(5, s.getCMarks());
            ps.executeUpdate();
        }
    }

    public List<Student> findAll() throws SQLException {
        String sql = "SELECT roll_no, name, java_marks, dbms_marks, c_marks FROM students ORDER BY roll_no";
        List<Student> list = new ArrayList<>();
        try (Connection c = connect(); Statement st = c.createStatement(); ResultSet rs = st.executeQuery(sql)) {
            while (rs.next()) {
                list.add(fromRow(rs));
            }
        }
        return list;
    }

    public Student findByRoll(String rollNo) throws SQLException {
        String sql = "SELECT roll_no, name, java_marks, dbms_marks, c_marks FROM students WHERE roll_no = ?";
        try (Connection c = connect(); PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setString(1, rollNo);
            try (ResultSet rs = ps.executeQuery()) {
                return rs.next() ? fromRow(rs) : null;
            }
        }
    }

    public Student topper() throws SQLException {
        String sql = "SELECT roll_no, name, java_marks, dbms_marks, c_marks FROM students "
                   + "ORDER BY (java_marks + dbms_marks + c_marks) DESC LIMIT 1";
        try (Connection c = connect(); Statement st = c.createStatement(); ResultSet rs = st.executeQuery(sql)) {
            return rs.next() ? fromRow(rs) : null;
        }
    }

    public boolean delete(String rollNo) throws SQLException {
        String sql = "DELETE FROM students WHERE roll_no = ?";
        try (Connection c = connect(); PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setString(1, rollNo);
            return ps.executeUpdate() > 0;
        }
    }

    private static Student fromRow(ResultSet rs) throws SQLException {
        return new Student(
            rs.getString("roll_no"),
            rs.getString("name"),
            rs.getInt("java_marks"),
            rs.getInt("dbms_marks"),
            rs.getInt("c_marks"));
    }
}

CREATE DATABASE IF NOT EXISTS grade_manager;
USE grade_manager;

CREATE TABLE IF NOT EXISTS students (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    roll_no     VARCHAR(20)  NOT NULL UNIQUE,
    name        VARCHAR(80)  NOT NULL,
    java_marks  INT NOT NULL CHECK (java_marks BETWEEN 0 AND 100),
    dbms_marks  INT NOT NULL CHECK (dbms_marks BETWEEN 0 AND 100),
    c_marks     INT NOT NULL CHECK (c_marks BETWEEN 0 AND 100),
    created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Every student with total and percentage
SELECT roll_no, name,
       java_marks + dbms_marks + c_marks                   AS total,
       ROUND((java_marks + dbms_marks + c_marks) / 3, 2)    AS percentage
FROM students
ORDER BY total DESC;

-- Class topper
SELECT name, java_marks + dbms_marks + c_marks AS total
FROM students
ORDER BY total DESC
LIMIT 1;

-- Class average per subject
SELECT ROUND(AVG(java_marks), 1) AS avg_java,
       ROUND(AVG(dbms_marks), 1) AS avg_dbms,
       ROUND(AVG(c_marks), 1)    AS avg_c
FROM students;

-- Students who failed at least one subject (below 35)
SELECT roll_no, name
FROM students
WHERE java_marks < 35 OR dbms_marks < 35 OR c_marks < 35;
